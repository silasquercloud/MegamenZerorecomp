cmake_minimum_required(VERSION 3.22)
project(MegaManZeroNative C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT ANDROID)
    message(FATAL_ERROR "Use android/gradlew para o APK; tools/check-host.sh para testes Linux.")
endif()
set(GBARECOMP_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/gbarecomp" CACHE PATH "Pinned runtime")
set(GBARECOMP_RUNTIME_UI_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/MegaManZeroRecomp/recomp-ui" CACHE PATH "Shared in-game settings")
set(GBARECOMP_GENERATED_BIOS_DIR "${CMAKE_CURRENT_SOURCE_DIR}/android/generated/bios" CACHE PATH "Local BIOS code")
set(GBARECOMP_SELFHEAL_RECOMPILE_DEFAULT OFF CACHE BOOL "No compiler on phones" FORCE)
set(GBARECOMP_ENABLE_MODS ON CACHE BOOL "Upstream game hooks" FORCE)
file(GLOB GAME_CODE CONFIGURE_DEPENDS "android/generated/game/*.cpp" "android/generated/game/*.c")
if(NOT GAME_CODE OR NOT EXISTS "${GBARECOMP_GENERATED_BIOS_DIR}/bios_recompiled.cpp")
    message(FATAL_ERROR "Faltam fontes nativos gerados. Execute tools/generate.py com sua ROM e BIOS verificadas. Não será produzido APK vazio.")
endif()
add_subdirectory("${GBARECOMP_ROOT}" engine EXCLUDE_FROM_ALL)
include(android/native/instrument-runtime.cmake)
set(GAME_SRC "${CMAKE_CURRENT_SOURCE_DIR}/MegaManZeroRecomp/src")
add_library(main SHARED android/native/entry.cpp android/native/enhanced.cpp android/native/vulkan/presenter.cpp android/native/vulkan/android_bridge.cpp "${GAME_SRC}/main.cpp"
    "${GAME_SRC}/mmz_extended_view.cpp" "${GAME_SRC}/mods/mmz_adaptive_view_plugin.cpp" ${GAME_CODE})
set_source_files_properties("${GAME_SRC}/main.cpp" PROPERTIES COMPILE_DEFINITIONS "main=mmz_game_main")
set_source_files_properties(${GAME_CODE} PROPERTIES COMPILE_OPTIONS "-O2;-g0")
target_include_directories(main PRIVATE "${GAME_SRC}" "${GBARECOMP_ROOT}/src/debug" "${GBARECOMP_ROOT}/src/gba" "${GBARECOMP_ROOT}/src/runtime" "${GBARECOMP_ROOT}/src/armv4t")
target_link_libraries(main PRIVATE "-Wl,--start-group" gbarecomp_runtime gbarecomp_debug gbarecomp_gba gbarecomp_armv4t "-Wl,--end-group" SDL2::SDL2)
# The runtime owns SDL_Renderer2 glue; recomp-ui supplies the single ImGui
# core, SDL platform backend and settings model through its supported helper.
include("${GBARECOMP_RUNTIME_UI_ROOT}/recomp_ui.cmake")
add_library(mmz_runtime_ui STATIC)
set_target_properties(mmz_runtime_ui PROPERTIES POSITION_INDEPENDENT_CODE ON)
recomp_target_launcher_ui(mmz_runtime_ui CONSOLE gba)
target_link_libraries(mmz_runtime_ui PUBLIC SDL2::SDL2)
target_link_libraries(main PRIVATE mmz_runtime_ui vulkan android)
target_compile_definitions(main PRIVATE GBARECOMP_DEFAULT_DEBUG_PORT=19862 GBARECOMP_WINDOW_TITLE="Mega Man Zero Native" GBARECOMP_DEFAULT_GAME_CONFIG="variants/megaman_zero/game.toml")
target_link_options(main PRIVATE "-Wl,--no-undefined" "-Wl,-z,max-page-size=16384")
